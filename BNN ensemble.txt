class ModelLoader:
    """Handles loading of pre-trained models and scalers"""
    
    def __init__(self, config):
        self.config = config
        self.models = []
        self.scalers = {}
        self.model_status = []
        self.scaler_status = {}
        
    def load_models_and_scalers(self):
        """Load all pre-trained models and scalers"""
        print("\n" + "=" * 60)
        print("LOADING PRE-TRAINED BNN MODELS AND SCALERS")
        print("=" * 60)
        
        # Define custom objects for model loading
        custom_objects = {
            'mse': 'mse',
            'mean_squared_error': 'mse',
            'mae': 'mae',
            'mean_absolute_error': 'mae'
        }
        
        # Load scalers first
        try:
            self.scalers['X'] = joblib.load(self.config.scaler_x_path)
            self.scalers['y'] = joblib.load(self.config.scaler_y_path)
            self.scaler_status = {'X': 'loaded', 'y': 'loaded'}
            print(f"✓ Scaler_X loaded from: {self.config.scaler_x_path}")
            print(f"✓ Scaler_y loaded from: {self.config.scaler_y_path}")
        except Exception as e:
            print(f"✗ Error loading scalers: {str(e)}")
            self.scaler_status = {'X': 'failed', 'y': 'failed'}
            raise
        
        # Load each model
        for i, model_path in enumerate(self.config.model_paths):
            try:
                if not os.path.exists(model_path):
                    print(f"✗ Model {i+1} not found: {model_path}")
                    self.model_status.append('not_found')
                    continue
                
                # Try loading with custom objects
                try:
                    model = models.load_model(model_path, custom_objects=custom_objects, compile=False)
                except:
                    with keras.utils.custom_object_scope(custom_objects):
                        model = models.load_model(model_path, compile=False)
                
                # Recompile model
                model.compile(optimizer='adam', loss='mse', metrics=['mae', 'mse'])
                
                self.models.append(model)
                self.model_status.append('loaded')
                
                # Get number of neurons from first dense layer
                neurons = model.layers[1].units if len(model.layers) > 1 else "unknown"
                print(f"✓ Model {i+1} loaded: {os.path.basename(os.path.dirname(model_path))} ({neurons} neurons)")
                
            except Exception as e:
                print(f"✗ Error loading model {i+1}: {str(e)}")
                self.model_status.append('failed')
                
                # Try alternative loading method
                try:
                    with h5py.File(model_path, 'r') as f:
                        print(f"  File is valid H5 format")
                    model = models.load_model(model_path, compile=False)
                    model.compile(optimizer='adam', loss='mse', metrics=['mae'])
                    self.models.append(model)
                    self.model_status[-1] = 'loaded_alternative'
                    print(f"✓ Model {i+1} loaded with alternative method")
                except Exception as e2:
                    print(f"✗ Alternative loading also failed: {str(e2)}")
        
        print(f"\nTotal models loaded: {len(self.models)} out of {len(self.config.model_paths)}")
        return self.models, self.scalers, self.model_status, self.scaler_status

class CommitteeModel:
    """
    Committee model that combines multiple BNN models for ensemble predictions
    with uncertainty quantification
    """
    
    def __init__(self, models, scaler_x, scaler_y, feature_names, target_name):
        self.models = models
        self.scaler_x = scaler_x
        self.scaler_y = scaler_y
        self.feature_names = feature_names
        self.target_name = target_name
        self.n_models = len(models)
        self.model_weights = np.ones(self.n_models) / self.n_models  # Equal weights initially
        
    def predict(self, X, n_mc_samples=100, return_all=False):
        """
        Make predictions with uncertainty
        
        Args:
            X: Input features (n_samples, n_features) - already scaled
            n_mc_samples: Number of MC samples per model
            return_all: If True, return all predictions for detailed analysis
            
        Returns:
            mean_pred: Mean prediction (n_samples, 1)
            std_pred: Prediction uncertainty (n_samples, 1)
            (optional) all_predictions: All individual predictions
        """
        all_predictions = []
        
        for model in self.models:
            model_preds = []
            for _ in range(n_mc_samples):
                pred = model.predict(X, verbose=0)
                model_preds.append(pred)
            all_predictions.append(np.array(model_preds))
        
        # Combine all predictions: (n_models * n_samples, n_samples, 1)
        all_predictions = np.vstack(all_predictions)
        
        # Calculate statistics
        mean_pred = np.mean(all_predictions, axis=0)
        std_pred = np.std(all_predictions, axis=0)
        
        if return_all:
            return mean_pred, std_pred, all_predictions
        return mean_pred, std_pred
    
    def predict_with_original_scale(self, X_original, n_mc_samples=100, return_all=False):
        """Make predictions with original (unscaled) inputs and outputs"""
        X_scaled = self.scaler_x.transform(X_original)
        mean_pred_scaled, std_pred_scaled, *rest = self.predict(X_scaled, n_mc_samples, return_all)
        
        mean_pred = self.scaler_y.inverse_transform(mean_pred_scaled)
        std_pred = std_pred_scaled * self.scaler_y.scale_
        
        if return_all:
            all_preds_scaled = rest[0]
            all_preds = np.array([self.scaler_y.inverse_transform(p) for p in all_preds_scaled])
            return mean_pred, std_pred, all_preds
        return mean_pred, std_pred
    
    def get_model_agreement(self, X, n_mc_samples=50):
        """Calculate agreement between models (std of model means)"""
        model_means = []
        
        for model in self.models:
            preds = []
            for _ in range(n_mc_samples):
                preds.append(model.predict(X, verbose=0))
            model_means.append(np.mean(preds, axis=0))
        
        model_means = np.array(model_means)
        agreement = np.std(model_means, axis=0)
        
        return agreement
    
    def save(self, path):
        """Save the committee model (saves model paths and weights)"""
        model_info = {
            'n_models': self.n_models,
            'model_paths': [None] * self.n_models,  # Don't save actual paths
            'model_weights': self.model_weights.tolist(),
            'feature_names': self.feature_names,
            'target_name': self.target_name,
            'scaler_x_mean': self.scaler_x.mean_.tolist() if hasattr(self.scaler_x, 'mean_') else None,
            'scaler_x_scale': self.scaler_x.scale_.tolist() if hasattr(self.scaler_x, 'scale_') else None,
            'scaler_y_mean': self.scaler_y.mean_.tolist() if hasattr(self.scaler_y, 'mean_') else None,
            'scaler_y_scale': self.scaler_y.scale_.tolist() if hasattr(self.scaler_y, 'scale_') else None
        }
        
        with open(path, 'w') as f:
            json.dump(model_info, f, indent=4)
        print(f"✓ Committee model info saved to: {path}")
    
    def save_pickle(self, path):
        """
        Save the complete committee model as a pickle file
        This saves the entire object including the loaded Keras models
        """
        try:
            # Create a copy of the models list to avoid any reference issues
            models_copy = self.models.copy()
            
            # Create a dictionary with all necessary components
            committee_pickle = {
                'models': models_copy,
                'scaler_x': self.scaler_x,
                'scaler_y': self.scaler_y,
                'feature_names': self.feature_names,
                'target_name': self.target_name,
                'n_models': self.n_models,
                'model_weights': self.model_weights,
                'creation_date': datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            }
            
            # Save using pickle
            with open(path, 'wb') as f:
                pickle.dump(committee_pickle, f)
            
            print(f"✓ Complete committee model saved as pickle to: {path}")
            return True
            
        except Exception as e:
            print(f"✗ Error saving committee model as pickle: {str(e)}")
            return False

def load_data(config):
    """Load and preprocess data"""
    try:
        data = pd.read_csv(config.data_path)
        # Strip whitespace from column names
        data.columns = data.columns.str.strip()
        
        # Also strip feature names for comparison
        feature_names_stripped = [name.strip() for name in config.feature_names]
        target_name_stripped = config.target_name.strip()
        
        # Check if target exists with stripped name
        if target_name_stripped not in data.columns:
            print(f"Available columns: {list(data.columns)}")
            raise KeyError(f"Target column '{target_name_stripped}' not found in data")
        
        # Features and target
        X = data[feature_names_stripped]
        y = data[target_name_stripped].values.reshape(-1, 1)
        
        # Check for NaN values
        if X.isnull().any().any() or np.isnan(y).any():
            print("Warning: NaN values found in data. Dropping NaN rows...")
            mask = ~(X.isnull().any(axis=1) | pd.isna(y).flatten())
            X = X[mask]
            y = y[mask]
        
        print(f"Data loaded successfully: {len(X)} samples, {len(X.columns)} features")
        return X, y
    except Exception as e:
        print(f"Error loading data: {str(e)}")
        raise

def split_data(X, y, config):
    """Split data into train, validation, and test sets (70/15/15)"""
    try:
        # First split: separate test data
        X_train_val, X_test, y_train_val, y_test = train_test_split(
            X, y, test_size=config.test_ratio, 
            random_state=config.random_seed, shuffle=True
        )
        
        # Second split: separate validation from remaining data
        val_size_adj = config.val_ratio / (1 - config.test_ratio)
        X_train, X_val, y_train, y_val = train_test_split(
            X_train_val, y_train_val, test_size=val_size_adj, 
            random_state=config.random_seed, shuffle=True
        )
        
        print(f"\nData Split Results:")
        print(f"  Train samples: {len(X_train)} ({len(X_train)/len(X)*100:.1f}%)")
        print(f"  Validation samples: {len(X_val)} ({len(X_val)/len(X)*100:.1f}%)")
        print(f"  Test samples: {len(X_test)} ({len(X_test)/len(X)*100:.1f}%)")
        print(f"  Overall samples: {len(X)} (100%)")
        
        return X_train, X_val, X_test, y_train, y_val, y_test
    except Exception as e:
        print(f"Error splitting data: {str(e)}")
        raise

def evaluate_predictions(y_true, y_pred_mean, y_pred_std=None, set_name=''):
    """Evaluate predictions with uncertainty metrics"""
    metrics = {}
    
    try:
        # Basic metrics
        metrics['mse'] = mean_squared_error(y_true, y_pred_mean)
        metrics['rmse'] = np.sqrt(metrics['mse'])
        metrics['mae'] = mean_absolute_error(y_true, y_pred_mean)
        metrics['r2'] = r2_score(y_true, y_pred_mean)
        
        # Pearson correlation
        try:
            metrics['r'] = pearsonr(y_true.flatten(), y_pred_mean.flatten())[0]
        except:
            metrics['r'] = 0.0
        
        # Uncertainty metrics
        if y_pred_std is not None and len(y_pred_std) > 0:
            metrics['mean_std'] = float(np.mean(y_pred_std))
            metrics['std_of_std'] = float(np.std(y_pred_std))
            
            # Calibration metrics
            with np.errstate(divide='ignore', invalid='ignore'):
                z_scores = (y_true - y_pred_mean) / (y_pred_std + 1e-8)
                metrics['calibration_error'] = float(np.mean(np.abs(z_scores[~np.isnan(z_scores)])))
            
            # 95% prediction interval coverage
            lower_bound = y_pred_mean - 1.96 * y_pred_std
            upper_bound = y_pred_mean + 1.96 * y_pred_std
            coverage = np.mean((y_true >= lower_bound) & (y_true <= upper_bound))
            metrics['coverage_95'] = float(coverage)
            
            # Prediction interval width
            metrics['pi_width'] = float(np.mean(upper_bound - lower_bound))
        
        print(f"\n{set_name} Metrics:")
        print(f"  MSE: {metrics['mse']:.4f}")
        print(f"  RMSE: {metrics['rmse']:.4f}")
        print(f"  MAE: {metrics['mae']:.4f}")
        print(f"  R²: {metrics['r2']:.4f}")
        print(f"  Pearson R: {metrics.get('r', 0):.4f}")
        
        if y_pred_std is not None and len(y_pred_std) > 0:
            print(f"  Mean Uncertainty: {metrics.get('mean_std', 0):.4f}")
            print(f"  95% PI Coverage: {metrics.get('coverage_95', 0):.2%}")
            print(f"  Avg PI Width: {metrics.get('pi_width', 0):.4f}")
        
        return metrics
    except Exception as e:
        print(f"Error in evaluation: {str(e)}")
        return {
            'mse': float('inf'), 'rmse': float('inf'), 'mae': float('inf'),
            'r2': -float('inf'), 'r': 0.0, 'mean_std': 0.0,
            'coverage_95': 0.0, 'pi_width': 0.0
        }

def evaluate_overall_predictions(y_true_dict, y_pred_mean_dict, y_pred_std_dict):
    """Evaluate overall predictions (train+val+test combined)"""
    try:
        y_true_all = np.concatenate([
            y_true_dict['train'].flatten(),
            y_true_dict['val'].flatten(),
            y_true_dict['test'].flatten()
        ])
        
        y_pred_all = np.concatenate([
            y_pred_mean_dict['train'].flatten(),
            y_pred_mean_dict['val'].flatten(),
            y_pred_mean_dict['test'].flatten()
        ])
        
        y_pred_std_all = np.concatenate([
            y_pred_std_dict['train'].flatten(),
            y_pred_std_dict['val'].flatten(),
            y_pred_std_dict['test'].flatten()
        ])
        
        overall_metrics = evaluate_predictions(
            y_true_all.reshape(-1, 1),
            y_pred_all.reshape(-1, 1),
            y_pred_std_all.reshape(-1, 1),
            'Overall (Train+Val+Test)'
        )
        
        return overall_metrics, y_true_all, y_pred_all, y_pred_std_all
    except Exception as e:
        print(f"Error calculating overall metrics: {str(e)}")
        default = {'mse': float('inf'), 'rmse': float('inf'), 'r2': -float('inf'), 
                  'mean_std': 0.0, 'coverage_95': 0.0}
        return default, np.array([]), np.array([]), np.array([])

def plot_ensemble_regression_with_uncertainty(y_true_dict, y_pred_mean_dict, y_pred_std_dict,
                                            per_model_predictions, model_names,
                                            save_path, config, overall_metrics=None,
                                            y_true_all=None, y_pred_all=None, y_pred_std_all=None):
    """Create comprehensive regression plots with uncertainty for ensemble"""
    
    try:
        fig = plt.figure(figsize=(24, 20))
        
        # Colors
        colors = {'train': 'blue', 'val': 'green', 'test': 'red'}
        model_colors = ['orange', 'purple', 'brown']
        
        # 1. Combined regression plot with uncertainty bands
        ax1 = plt.subplot(3, 4, 1)
        
        # Determine global min/max
        global_min, global_max = float('inf'), float('-inf')
        for set_name in ['train', 'val', 'test']:
            y_true = y_true_dict[set_name]
            y_pred = y_pred_mean_dict[set_name]
            if len(y_true) > 0:
                global_min = min(global_min, np.min(y_true), np.min(y_pred))
                global_max = max(global_max, np.max(y_true), np.max(y_pred))
        
        if global_min == float('inf'):
            global_min, global_max = 0, 1
        
        padding = (global_max - global_min) * 0.1
        global_min -= padding
        global_max += padding
        
        # Plot each dataset
        for set_name in ['train', 'val', 'test']:
            y_true = y_true_dict[set_name]
            y_pred = y_pred_mean_dict[set_name]
            y_std = y_pred_std_dict[set_name]
            
            if len(y_true) == 0:
                continue
            
            # Sort for clean error bands
            sorted_idx = np.argsort(y_true.flatten())
            y_true_sorted = y_true.flatten()[sorted_idx]
            y_pred_sorted = y_pred.flatten()[sorted_idx]
            y_std_sorted = y_std.flatten()[sorted_idx]
            
            # Plot error bands
            ax1.fill_between(y_true_sorted,
                            y_pred_sorted - 2*y_std_sorted,
                            y_pred_sorted + 2*y_std_sorted,
                            alpha=0.2, color=colors[set_name])
            
            # Plot predictions with error bars
            subsample_step = max(1, len(y_true_sorted)//20)
            ax1.errorbar(y_true_sorted[::subsample_step],
                        y_pred_sorted[::subsample_step],
                        yerr=2*y_std_sorted[::subsample_step],
                        fmt='o', color=colors[set_name], alpha=0.7,
                        markersize=6, capsize=3, label=f'{set_name.capitalize()}')
        
        # Ideal line
        ax1.plot([global_min, global_max], [global_min, global_max], 'k--', linewidth=2, label='Ideal')
        
        ax1.set_xlim(global_min, global_max)
        ax1.set_ylim(global_min, global_max)
        ax1.set_xlabel('True Hydrogen Absorption Weight')
        ax1.set_ylabel('Predicted Hydrogen Absorption Weight')
        ax1.set_title('Ensemble BNN Regression with Uncertainty\n(All Data Combined)', fontweight='bold')
        ax1.legend(loc='best')
        ax1.grid(True, alpha=0.3)
        
        # Add metrics box
        if overall_metrics:
            textstr = '\n'.join((
                f'Overall R² = {overall_metrics.get("r2", 0):.4f}',
                f'Overall RMSE = {overall_metrics.get("rmse", 0):.4f}',
                f'Overall MAE = {overall_metrics.get("mae", 0):.4f}',
                f'Mean Uncertainty = {overall_metrics.get("mean_std", 0):.4f}',
                f'95% PI Coverage = {overall_metrics.get("coverage_95", 0):.2%}'))
        else:
            textstr = f'Ensemble of {len(model_names)} Models'

                           y_pred_sorted - 2*y_std_sorted,

            mae = mean_absolute_error(y_true, y_pred)

            ax.set_ylabel('Predicted Value')

            y_pred = y_pred_mean_dict[set_name]

        ax7.set_xlabel('Predicted Value')
        ax7.set_ylabel('Residual')

        for model_idx in range(len(per_model_predictions['test'])):
            model_preds = per_model_predictions['test'][model_idx]
            model_stds = per_model_predictions['test_std'][model_idx]
            if len(model_preds) > 0:
                lower = model_preds - 1.96 * model_stds
                upper = model_preds + 1.96 * model_stds
                coverage = np.mean((y_true_dict['test'] >= lower) & (y_true_dict['test'] <= upper))
                coverage_by_model.append(coverage)
            else:
                coverage_by_model.append(0)
        
        bars = ax11.bar(range(len(coverage_by_model)), coverage_by_model, 
                       color='steelblue', alpha=0.7)

        ax11.set_xlabel('Model Index')
        ax11.set_ylabel('Coverage Probability')
        ax11.set_title('95% PI Coverage by Model (Test Set)')
        ax11.set_xticks(range(len(coverage_by_model)))
        ax11.set_xticklabels([f'M{i+1}' for i in range(len(coverage_by_model))])
        ax11.set_ylim([0, 1])
        ax11.legend()
        ax11.grid(True, alpha=0.3, axis='y')
        
        # 12. Ensemble improvement
        ax12 = plt.subplot(3, 4, 12)
        
        # Calculate improvement over best individual model
        best_model_r2 = max(model_r2) if model_r2 else 0
        improvement = ensemble_r2 - best_model_r2
        
        metrics_improvement = ['R²', 'MAE', 'Uncertainty']
        improvement_values = [
            improvement,
            (np.min(model_mae) - overall_metrics.get('mae', 0)) if model_mae else 0,
            (np.min([np.mean(m) for m in per_model_predictions['test_std'] if len(m) > 0]) - overall_metrics.get('mean_std', 0)) if per_model_predictions['test_std'] else 0
        ]
        
        colors_imp = ['green' if v > 0 else 'red' for v in improvement_values]
        bars = ax12.bar(metrics_improvement, improvement_values, color=colors_imp, alpha=0.7)
        ax12.axhline(y=0, color='black', linewidth=1)
        ax12.set_ylabel('Improvement over Best Model')
        ax12.set_title('Ensemble Improvement')
        ax12.grid(True, alpha=0.3, axis='y')
        
        for bar, val in zip(bars, improvement_values):
            height = bar.get_height()
            ax12.text(bar.get_x() + bar.get_width()/2., height + 0.001 if height >= 0 else height - 0.01,
                     f'{val:.3f}', ha='center', va='bottom' if height >= 0 else 'top', fontsize=9)
        
        plt.suptitle(f'Ensemble BNN Model Analysis - Committee of {len(model_names)} Models\n70/15/15 Split with Overall Results', 
                    fontsize=16, fontweight='bold')
        plt.tight_layout()
        plt.savefig(os.path.join(save_path, 'ensemble_regression_analysis.png'), dpi=300, bbox_inches='tight')
        plt.close()
        
    except Exception as e:
        print(f"Error plotting ensemble regression: {str(e)}")

        features_original = feature_names

                print(f"  {i+1}. {row['Feature']:20s}: ST = {row['ST_Mean']:+.4f} ± {row['ST_Std']:.4f} ({row['ST_Effect']})")
            
            return combined_df, summary_df
        
    except Exception as e:
        print(f"Error creating ensemble Sobol global sensitivity plot: {str(e)}")
        return None, None

# SOBOL SENSITIVITY ANALYSIS
# ===============================
def ensemble_sobol_sensitivity_analysis(models, scaler_X, scaler_y, feature_names, 
                                        X_train, save_path, n_samples=500):
    """
    Sobol sensitivity analysis for ensemble model
    """
    if not SOBOL_AVAILABLE:
        print("SALib not available. Skipping Sobol analysis.")
        return None
    
    print(f"\nPerforming Ensemble Sobol Sensitivity Analysis...")
    
    try:
        # Define bounds based on training data
        bounds = []
        for col in feature_names:
            lower = float(np.min(X_train[col]))
            upper = float(np.max(X_train[col]))
            bounds.append([lower, upper])
        
        problem = {
            'num_vars': len(feature_names),
            'names': list(feature_names),
            'bounds': bounds
        }
        
        # Generate Sobol samples
        param_values = saltelli.sample(problem, n_samples, calc_second_order=True)
        
        # Scale inputs
        param_values_scaled = scaler_X.transform(param_values)
        
        # Get ensemble predictions with MC Dropout
        all_predictions = []
        for model in models:
            model_preds = []
            for _ in range(min(20, 50)):  # MC samples per model
                pred_scaled = model.predict(param_values_scaled, verbose=0)
                model_preds.append(pred_scaled)
            all_predictions.append(np.array(model_preds))
        
        # Combine all predictions
        all_predictions = np.vstack(all_predictions)
        Y_pred_mean = np.mean(all_predictions, axis=0)
        Y_pred = scaler_y.inverse_transform(Y_pred_mean).flatten()
        
        # Perform Sobol analysis
        sobol_indices = sobol.analyze(problem, Y_pred, print_to_console=False, 
                                       calc_second_order=True)
        
        # Calculate signed direction
        signed_corr = []
        for i in range(len(feature_names)):
            corr, _ = pearsonr(param_values[:, i], Y_pred)
            signed_corr.append(corr if not np.isnan(corr) else 0)
        
        # Apply sign to indices
        s1_signed = np.sign(signed_corr) * sobol_indices['S1']
        st_signed = np.sign(signed_corr) * sobol_indices['ST']
        
        # Collect results
        sobol_results = []
        for i in range(len(feature_names)):
            sobol_results.append({
                'Feature': feature_names[i],
                'Signed_S1': s1_signed[i],
                'S1_Conf': sobol_indices['S1_conf'][i],
                'Signed_ST': st_signed[i],
                'ST_Conf': sobol_indices['ST_conf'][i],
                'Correlation': signed_corr[i],
                'S1_abs': np.abs(s1_signed[i]),
                'ST_abs': np.abs(st_signed[i])
            })
        
        sobol_df = pd.DataFrame(sobol_results)
        sobol_df['S1_Rank'] = sobol_df['S1_abs'].rank(ascending=False).astype(int)
        sobol_df['ST_Rank'] = sobol_df['ST_abs'].rank(ascending=False).astype(int)
        
        # Save results
        sobol_df.to_csv(os.path.join(save_path, 'ensemble_sobol_sensitivity.csv'), index=False)
        
        # Create visualization
        fig, axes = plt.subplots(1, 2, figsize=(14, 6))
        
        # Signed S1
        y_pos = np.arange(len(sobol_df))
        axes[0].barh(y_pos, sobol_df['Signed_S1'], 
                     xerr=sobol_df['S1_Conf'],
                     color=['crimson' if x < 0 else 'teal' for x in sobol_df['Signed_S1']],
                     capsize=3)
        axes[0].axvline(0, color='black', linestyle='--')
        axes[0].set_yticks(y_pos)
        axes[0].set_yticklabels(sobol_df['Feature'])
        axes[0].set_xlabel('Signed First-order Sobol Index (S1)')
        axes[0].set_title('Ensemble First-order Sensitivity')
        
        # Signed ST
        axes[1].barh(y_pos, sobol_df['Signed_ST'],
                     xerr=sobol_df['ST_Conf'],
                     color=['crimson' if x < 0 else 'teal' for x in sobol_df['Signed_ST']],
                     capsize=3)
        axes[1].axvline(0, color='black', linestyle='--')
        axes[1].set_yticks(y_pos)
        axes[1].set_yticklabels(sobol_df['Feature'])
        axes[1].set_xlabel('Signed Total-order Sobol Index (ST)')
        axes[1].set_title('Ensemble Total-order Sensitivity')
        
        plt.suptitle('Ensemble Sobol Sensitivity Analysis', fontweight='bold')
        plt.tight_layout()
        plt.savefig(os.path.join(save_path, 'ensemble_sobol_sensitivity.png'), dpi=300, bbox_inches='tight')
        plt.close()
        
        return sobol_results
        
    except Exception as e:
        print(f"Warning: Ensemble Sobol analysis failed: {str(e)}")
        return None

# MONTE CARLO PREDICTION FUNCTION
# ===============================
def monte_carlo_predict_ensemble(models, X, n_samples=100):
    """
    Make predictions using ensemble of BNN models with MC Dropout
    Returns mean and std across both ensemble members and MC samples
    """
    try:
        all_predictions = []  # Will store predictions from all models and MC samples
        
        for model_idx, model in enumerate(models):
            model_predictions = []
            
            for _ in range(n_samples):
                # Each forward pass samples different dropout masks
                pred = model.predict(X, verbose=0)
                model_predictions.append(pred)
            
            model_predictions = np.array(model_predictions)  # (n_samples, n_data, 1)
            all_predictions.append(model_predictions)
        
        # Combine all predictions: (n_models * n_samples, n_data, 1)
        all_predictions = np.vstack(all_predictions)
        
        # Calculate statistics
        mean_pred = np.mean(all_predictions, axis=0)
        std_pred = np.std(all_predictions, axis=0)
        
        # Also calculate per-model statistics for analysis
        per_model_means = []
        per_model_stds = []
        for model_idx, model in enumerate(models):
            model_preds = all_predictions[model_idx*n_samples:(model_idx+1)*n_samples]
            per_model_means.append(np.mean(model_preds, axis=0))
            per_model_stds.append(np.std(model_preds, axis=0))
        
        per_model_means = np.array(per_model_means)  # (n_models, n_data, 1)
        per_model_stds = np.array(per_model_stds)    # (n_models, n_data, 1)
        
        return mean_pred, std_pred, all_predictions, per_model_means, per_model_stds
    except Exception as e:
        print(f"Error in ensemble MC prediction: {str(e)}")
        # Return zeros as fallback
        if len(models) > 0:
            mean_pred = models[0].predict(X, verbose=0)
        else:
            mean_pred = np.zeros((X.shape[0], 1))
        std_pred = np.zeros_like(mean_pred)
        return mean_pred, std_pred, np.array([mean_pred]), np.array([mean_pred]), np.array([std_pred])
