class UniversalCommitteeModel:
    """Universal committee model that handles different model types"""
    
    def __init__(self, models, scaler_x, scaler_y):
        self.models = models
        self.scaler_x = scaler_x
        self.scaler_y = scaler_y
        self.model_types = [type(model).__name__ for model in models]
        
    def predict(self, X, batch_size=100, verbose=True):
        """Make predictions with batch processing and error handling"""
        # Scale input
        try:
            X_scaled = self.scaler_x.transform(X)
        except Exception as e:
            print(f"Error scaling input: {e}")
            raise
            
        # Process in batches
        n_samples = len(X_scaled)
        predictions = []
        
        for i in range(0, n_samples, batch_size):
            batch = X_scaled[i:min(i+batch_size, n_samples)]
            batch_preds = []
            
            for model_idx, model in enumerate(self.models):
                try:
                    pred = model.predict(batch)
                    # Handle different output shapes
                    if hasattr(pred, 'shape'):
                        if len(pred.shape) == 1:
                            pred = pred.reshape(-1, 1)
                    else:
                        pred = np.array(pred).reshape(-1, 1)
                    batch_preds.append(pred)
                except Exception as e:
                    print(f"Error with model {model_idx} on batch {i}: {e}")
                    # Use mean of other models as fallback
                    if batch_preds:
                        pred = np.mean(batch_preds, axis=0)
                    else:
                        pred = np.zeros((len(batch), 1))
                    batch_preds.append(pred)
            
            # Average predictions for this batch
            batch_avg = np.mean(batch_preds, axis=0)
            predictions.append(batch_avg)
            
            if verbose and (i + batch_size) % (batch_size * 5) == 0:
                print(f"  Processed {min(i+batch_size, n_samples)}/{n_samples} samples")
        
        # Combine batches
        y_pred_scaled = np.vstack(predictions)
        
        # Inverse transform
        try:
            return self.scaler_y.inverse_transform(y_pred_scaled)
        except Exception as e:
            print(f"Error in inverse transform: {e}")
            return y_pred_scaled

def safe_mkdir(path):
    """Safely create directory"""
    try:
        os.makedirs(path, exist_ok=True)
        return True
    except Exception as e:
        print(f"Error creating directory {path}: {e}")
        return False

def load_model_safe(path):
    """Safely load a model with error handling"""
    try:
        model = joblib.load(path)
        print(f"  ✓ Loaded: {os.path.basename(path)}")
        return model
    except Exception as e:
        print(f"  ✗ Failed to load {os.path.basename(path)}: {e}")
        return None

def evaluate_metrics(y_true, y_pred):
    """Calculate all metrics safely"""
    y_true = np.array(y_true).flatten()
    y_pred = np.array(y_pred).flatten()
    
    mse = mean_squared_error(y_true, y_pred)
    rmse = np.sqrt(mse)
    mae = mean_absolute_error(y_true, y_pred)
    r2 = r2_score(y_true, y_pred) * 100
    
    # Safe accuracy calculation
    mean_true = np.mean(y_true)
    if mean_true != 0:
        acc = (1 - (mae / mean_true)) * 100
    else:
        acc = 0
        
    return {
        'MSE': mse,
        'RMSE': rmse,
        'MAE': mae,
        'R2 (%)': r2,
        'Accuracy (%)': acc
    }

def analyze_individual_models(committee, X, Y):

    print("\n" + "=" * 60)
    print("INDIVIDUAL MODEL PERFORMANCE ANALYSIS")
    print("=" * 60)
    
    results = []
    
    for idx, model in enumerate(committee.models):
        print(f"\nAnalyzing Model {idx + 1}...")
        try:
            # Scale input
            X_scaled = committee.scaler_x.transform(X)
            
            # Predict with individual model
            Y_pred_scaled = model.predict(X_scaled)
            
            # Handle different output shapes
            if len(Y_pred_scaled.shape) == 1:
                Y_pred_scaled = Y_pred_scaled.reshape(-1, 1)
            
            # Inverse transform
            Y_pred = committee.scaler_y.inverse_transform(Y_pred_scaled)
            
            # Calculate metrics
            metrics = evaluate_metrics(Y, Y_pred)
            metrics['Model_Number'] = idx + 1
            metrics['Model_Type'] = committee.model_types[idx] if idx < len(committee.model_types) else 'Unknown'
            results.append(metrics)
            
            print(f"  R²: {metrics['R2 (%)']:.2f}%, RMSE: {metrics['RMSE']:.4f}")
