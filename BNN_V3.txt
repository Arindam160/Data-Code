# ---------------- DEFINE BNN MODEL ----------------
def build_bnn(input_dim, hidden_units, dropout_rate=0.05):
    inputs = layers.Input(shape=(input_dim,))
    x = layers.Dense(hidden_units, activation="relu")(inputs)
    x = layers.Dropout(dropout_rate)(x, training=True)  # MC Dropout active
    x = layers.Dense(hidden_units, activation="relu")(x)
    x = layers.Dropout(dropout_rate)(x, training=True)
    outputs = layers.Dense(1)(x)

    model = models.Model(inputs, outputs)
    model.compile(optimizer=optimizers.Adam(learning_rate=0.001), loss="mse")
    return model

# ---------------- METRICS ----------------
def evaluate(y_true, y_pred):
    mse = mean_squared_error(y_true, y_pred)
    rmse = np.sqrt(mse)
    mae = mean_absolute_error(y_true, y_pred)
    r2 = r2_score(y_true, y_pred) * 100
    acc = (1 - (mae / np.mean(y_true))) * 100
    return mse, rmse, mae, r2, acc

# ---------------- CUSTOM CALLBACK FOR EPOCH DATA ----------------
class EpochDataLogger(callbacks.Callback):
    def __init__(self, run_path, X_train, Y_train, X_test, Y_test, scaler_Y):
        super().__init__()
        self.run_path = run_path
        self.X_train = X_train
        self.Y_train = Y_train
        self.X_test = X_test
        self.Y_test = Y_test
        self.scaler_Y = scaler_Y
        self.epoch_data = []
        
    def on_epoch_end(self, epoch, logs=None):
        # Predict with MC Dropout active
        Y_train_pred = self.model.predict(self.X_train, verbose=0)
        Y_test_pred = self.model.predict(self.X_test, verbose=0)
        
        # Inverse transform
        Y_train_pred = self.scaler_Y.inverse_transform(Y_train_pred)
        Y_test_pred = self.scaler_Y.inverse_transform(Y_test_pred)
        Y_train_orig = self.scaler_Y.inverse_transform(self.Y_train)
        Y_test_orig = self.scaler_Y.inverse_transform(self.Y_test)
        
        # Calculate metrics
        train_metrics = evaluate(Y_train_orig, Y_train_pred)
        test_metrics = evaluate(Y_test_orig, Y_test_pred)
        
        # Store epoch data
        self.epoch_data.append({
            'epoch': epoch + 1,
            'train_loss': logs.get('loss', np.nan),
            'val_loss': logs.get('val_loss', np.nan),
            'train_mse': train_metrics[0],
            'train_rmse': train_metrics[1],
            'train_mae': train_metrics[2],
            'train_r2': train_metrics[3],
            'train_acc': train_metrics[4],
            'test_mse': test_metrics[0],
            'test_rmse': test_metrics[1],
            'test_mae': test_metrics[2],
            'test_r2': test_metrics[3],
            'test_acc': test_metrics[4]
        })
        
        # Save epoch data every 10 epochs
        if (epoch + 1) % 10 == 0:
            epoch_df = pd.DataFrame(self.epoch_data)
            epoch_df.to_csv(os.path.join(self.run_path, "epoch_data.csv"), index=False)
    
    def on_train_end(self, logs=None):
        # Save final epoch data
        epoch_df = pd.DataFrame(self.epoch_data)
        epoch_df.to_csv(os.path.join(self.run_path, "epoch_data.csv"), index=False)

# ---------------- TRAIN FUNCTION ----------------
def train_bnn(run_id, run_path, X_train_scaled, Y_train_scaled, X_test_scaled, Y_test_scaled, 
              scaler_Y, hidden_units, epochs=300, patience=30):
    model = build_bnn(input_dim=X_train_scaled.shape[1], hidden_units=hidden_units)

    # Create epoch data logger
    epoch_logger = EpochDataLogger(run_path, X_train_scaled, Y_train_scaled, 
                                  X_test_scaled, Y_test_scaled, scaler_Y)
    
    es = callbacks.EarlyStopping(
        monitor="val_loss", patience=patience, restore_best_weights=True, verbose=1
    )

    history = model.fit(
        X_train_scaled, Y_train_scaled,
        validation_data=(X_test_scaled, Y_test_scaled),
        epochs=epochs,
        batch_size=32,
        verbose=0,
        callbacks=[es, epoch_logger]
    )

    # Predictions with MC Dropout active for uncertainty
    Y_train_pred = model.predict(X_train_scaled, verbose=0)
    Y_test_pred = model.predict(X_test_scaled, verbose=0)

    # Inverse transform
    Y_train_pred = scaler_Y.inverse_transform(Y_train_pred)
    Y_test_pred = scaler_Y.inverse_transform(Y_test_pred)
    Y_train_orig = scaler_Y.inverse_transform(Y_train_scaled)
    Y_test_orig = scaler_Y.inverse_transform(Y_test_scaled)

    # Metrics
    train_metrics = evaluate(Y_train_orig, Y_train_pred)
    test_metrics = evaluate(Y_test_orig, Y_test_pred)
    
    # Overall metrics (train + test combined)
    Y_all_orig = np.vstack([Y_train_orig, Y_test_orig])
    Y_all_pred = np.vstack([Y_train_pred, Y_test_pred])
    overall_metrics = evaluate(Y_all_orig, Y_all_pred)

    print(f"\n=== Run {run_id} Results ===")
    print(f"Train -> MSE: {train_metrics[0]:.4f}, RMSE: {train_metrics[1]:.4f}, "
          f"MAE: {train_metrics[2]:.4f}, R²: {train_metrics[3]:.2f}%, Acc: {train_metrics[4]:.2f}%")
    print(f"Test  -> MSE: {test_metrics[0]:.4f}, RMSE: {test_metrics[1]:.4f}, "
          f"MAE: {test_metrics[2]:.4f}, R²: {test_metrics[3]:.2f}%, Acc: {test_metrics[4]:.2f}%")
    print(f"Overall -> MSE: {overall_metrics[0]:.4f}, RMSE: {overall_metrics[1]:.4f}, "
          f"MAE: {overall_metrics[2]:.4f}, R²: {overall_metrics[3]:.2f}%, Acc: {overall_metrics[4]:.2f}%")

    # Training curve with early stop
    plt.figure(figsize=(8, 6))
    plt.plot(history.history["loss"], label="Train MSE", linewidth=2)
    plt.plot(history.history["val_loss"], label="Test MSE", linewidth=2)
    best_epoch = np.argmin(history.history["val_loss"])
    plt.axvline(best_epoch, color="red", linestyle="--", label=f"Early Stop @ {best_epoch+1}")
    plt.xlabel("Epochs")
    plt.ylabel("MSE")
    plt.title(f"BNN Run {run_id} - MSE vs Epoch ({hidden_units} neurons)")
    plt.legend()
    plt.tight_layout()
    plt.savefig(os.path.join(run_path, f"BNN_Run_{run_id}_MSE.png"), dpi=300)
    plt.close()

    # ======================
    # Combined Regression Plot (Train + Test = Overall)
    # ======================
    plt.figure(figsize=(8, 6))
    plt.scatter(Y_all_orig, Y_all_pred, edgecolors="black", facecolors="teal",
                alpha=0.75, s=70, label="All Data (Train + Test)")
    min_val = float(min(Y_all_orig))
    max_val = float(max(Y_all_orig))
    plt.plot([min_val, max_val], [min_val, max_val], "r--", lw=2, label="Ideal")
    plt.xlabel("Actual Hydrogen Des. Time")
    plt.ylabel("Predicted Hydrogen Des. Time")
    plt.title(f"BNN Run {run_id} - Regression Plot (Overall Data, {hidden_units} neurons)")
    plt.legend()
    plt.tight_layout()
    plt.savefig(os.path.join(run_path, f"BNN_Run_{run_id}_Regression_Overall.png"), dpi=300)
    plt.close()

    # Save predictions for this run
    pred_df = pd.DataFrame({
        'Actual': Y_all_orig.flatten(),
        'Predicted': Y_all_pred.flatten(),
        'Set': ['Train'] * len(Y_train_orig) + ['Test'] * len(Y_test_orig)
    })
    pred_df.to_csv(os.path.join(run_path, f"BNN_Run_{run_id}_Predictions.csv"), index=False)

    # Return metrics and model
    return model, test_metrics[0], train_metrics, test_metrics, overall_metrics, epoch_logger.epoch_data

# ---------------- SOBOL ANALYSIS FUNCTION ----------------
def perform_sobol_analysis(run_id, run_path, model, scaler_X, scaler_Y, X):
    # Define the problem (safe bounds)
    bounds = []
    for col in X.columns:
        col_min, col_max = X[col].min(), X[col].max()
        if col_min == col_max:  
            # avoid zero-width bound
            col_min -= 1e-6
            col_max += 1e-6
        bounds.append([col_min, col_max])

    problem = {
        'num_vars': X.shape[1],
        'names': X.columns.tolist(),
        'bounds': bounds
    }

    # Sobol sampling
    param_values = saltelli.sample(problem, 1024, calc_second_order=True)

    # Ensure shape matches before scaling
    if param_values.shape[1] != X.shape[1]:
        raise ValueError(
            f"Shape mismatch: Sobol sample has {param_values.shape[1]} cols, "
            f"but dataset has {X.shape[1]} features"
        )

    # Scale params into model space
    param_values_scaled = scaler_X.transform(param_values)

    # Predict using trained model
    Y_sobol_pred_scaled = model.predict(param_values_scaled, verbose=0)
    Y_sobol_pred = scaler_Y.inverse_transform(Y_sobol_pred_scaled).flatten()

    # Sobol Analysis
    sobol_indices = sobol.analyze(problem, Y_sobol_pred, print_to_console=False)

    # Signed direction via Pearson correlation (approx directional sensitivity)
    signed_corr = [
        pearsonr(param_values[:, i], Y_sobol_pred)[0] for i in range(X.shape[1])
    ]

    # Apply sign to indices
    s1_signed = np.sign(signed_corr) * sobol_indices['S1']
    st_signed = np.sign(signed_corr) * sobol_indices['ST']

    # Collect results
    sobol_df_signed = pd.DataFrame({
        'Feature': problem['names'],
        'Signed S1': s1_signed,
        'S1 Conf': sobol_indices['S1_conf'],
        'Signed ST': st_signed,
        'ST Conf': sobol_indices['ST_conf'],
        'Correlation': signed_corr
    })

    sobol_df_signed['Original Order'] = range(len(sobol_df_signed))
    sobol_df_signed.sort_values(by='Original Order', inplace=True)
    sobol_df_signed.drop(columns='Original Order', inplace=True)

    print(f"\n--- Run {run_id} Signed Sobol Sensitivity Analysis Results (BNN) ---")
    print(sobol_df_signed)

    # Save
    signed_csv_path = os.path.join(run_path, "Signed_Sobol_Sensitivity.csv")
    sobol_df_signed.to_csv(signed_csv_path, index=False)
    print(f"\nSigned Sobol results saved at: {signed_csv_path}")

    # Plotting
    plt.rcParams.update({'font.size': 12, 'figure.figsize': (12, 6)})
    sns.set_style("white")

    # --- Plot Signed S1 ---
    plt.figure()
    colors = ['crimson' if val < 0 else 'teal' for val in sobol_df_signed['Signed S1']]
    plt.barh(sobol_df_signed['Feature'], sobol_df_signed['Signed S1'], color=colors)
    plt.axvline(0, color='black', linestyle='--')
    plt.title(f"Run {run_id} - Signed First-order Sobol Sensitivity (1st order)")
    plt.xlabel("Signed S1")
    plt.ylabel("Feature")
    plt.tight_layout()
    plt.savefig(os.path.join(run_path, "Signed_Sobol_S1.png"))
    plt.close()

    # --- Plot Signed ST ---
    plt.figure()
    colors = ['crimson' if val < 0 else 'teal' for val in sobol_df_signed['Signed ST']]
    plt.barh(sobol_df_signed['Feature'], sobol_df_signed['Signed ST'], color=colors)
    plt.axvline(0, color='black', linestyle='--')
    plt.title(f"Run {run_id} - Signed Total-order Sobol Sensitivity")
    plt.xlabel("Signed ST")
    plt.ylabel("Feature")
    plt.tight_layout()
    plt.savefig(os.path.join(run_path, "Signed_Sobol.png"))
    plt.close()
    
    return sobol_df_signed
