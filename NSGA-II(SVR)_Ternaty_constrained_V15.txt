def get_density_from_atomic_weight(aw: float) -> float:
    """Density based on atomic weight - piecewise function from experimental data"""
    if 80 < aw <= 90:
        return 4.47      # Yttrium
    elif 130 < aw <= 140:
        return 6.15      # Lanthanum
    elif 140 < aw <= 141:
        return 6.77      # Cerium/Praseodymium
    elif 150 < aw <= 151:
        return 7.52      # Samarium
    else:
        if 90 < aw <= 130:
            return 4.47 + (6.15 - 4.47) * (aw - 90) / 40
        elif 141 < aw <= 150:
            return 6.77 + (7.52 - 6.77) * (aw - 141) / 9
        else:
            return max(4.0, min(8.0, 6.0))

def calculate_theoretical_radius(aw: float) -> float:
    """Calculate theoretical atomic radius from atomic weight and density"""
    density = get_density_from_atomic_weight(aw)
    volume_per_atom = aw / (density * N_A)
    return ((3 * volume_per_atom) / (4 * np.pi)) ** (1/3) * CM_TO_PM

def calculate_constraint_violation(actual: float, theoretical: float, tolerance: float) -> float:
    """
    Calculate constraint violation: max(0, |actual/theoretical - 1| - tolerance)
    Returns 0 if within tolerance, positive violation otherwise
    """
    if abs(theoretical) < 1e-10:
        return 0.0
    deviation = abs(actual / theoretical - 1)
    return max(0.0, deviation - tolerance)

def constraint_structural_violation(X: np.ndarray, tolerance: float = 0.25) -> np.ndarray:
    """C1: Atomic radius constraint violation"""
    violations = np.zeros(len(X))
    aw_idx = VAR_NAMES.index("M2_AW")
    ar_idx = VAR_NAMES.index("M2_AR")
    
    for i in range(len(X)):
        theoretical = calculate_theoretical_radius(X[i, aw_idx])
        actual = X[i, ar_idx]
        violations[i] = calculate_constraint_violation(actual, theoretical, tolerance)
    return violations

def constraint_pauling_violation(X: np.ndarray, tolerance: float = 0.25) -> np.ndarray:
    """C2: Pauling electronegativity constraint violation"""
    violations = np.zeros(len(X))
    en_idx = VAR_NAMES.index("M2_EN")
    enthal_idx = VAR_NAMES.index("M2_Enthal")
    
    for i in range(len(X)):
        theoretical = -PAULING_K * (X[i, en_idx] ** 2)
        actual = X[i, enthal_idx]
        violations[i] = calculate_constraint_violation(actual, theoretical, tolerance)
    return violations

def constraint_cohesive_violation(X: np.ndarray, tolerance: float = 0.25) -> np.ndarray:
    """C3: Cohesive energy constraint violation"""
    violations = np.zeros(len(X))
    ar_idx = VAR_NAMES.index("M2_AR")
    ecoh_idx = VAR_NAMES.index("M2_Ecoh")
    
    for i in range(len(X)):
        theoretical = REF_E * (REF_R / X[i, ar_idx]) ** COHESIVE_EXPONENT
        actual = X[i, ecoh_idx]
        violations[i] = calculate_constraint_violation(actual, theoretical, tolerance)
    return violations

def total_constraint_violation(X: np.ndarray, tolerance: float = 0.25) -> np.ndarray:
    """Sum of all constraint violations for constraint dominance"""
    v1 = constraint_structural_violation(X, tolerance)
    v2 = constraint_pauling_violation(X, tolerance)
    v3 = constraint_cohesive_violation(X, tolerance)
    return v1 + v2 + v3

def generate_feasible_population(n_individuals: int, scenario_name: str = "full") -> np.ndarray:
    """Generate population using scientifically valid starting points
    
    Args:
        n_individuals: Number of individuals to generate
        scenario_name: One of "full", "Absorption_Only", "Desorption_Only", 
                      "AbsTime_vs_DesTime", "AbsWt_vs_DesWt"
    """
    np.random.seed(SEED_BASE)
    population = []
    
    # Create known valid material compositions from REFERENCES
    known_materials = []
    for material, props in MATERIAL_REFERENCES.items():
        # Build a complete individual with all variables
        individual = np.zeros(len(VAR_NAMES))
        individual[VAR_NAMES.index("M2_AW")] = props["M2_AW"]
        individual[VAR_NAMES.index("M2_EN")] = props["M2_EN"]
        individual[VAR_NAMES.index("M2_AR")] = props["M2_AR"]
        individual[VAR_NAMES.index("M2_Enthal")] = props["M2_Enthal"]
        individual[VAR_NAMES.index("M2_Ecoh")] = props["M2_Ecoh"]
        # Set default values for other variables (mid-range)
        individual[VAR_NAMES.index("Ni_atom%")] = (BOUNDS["Ni_atom%"][0] + BOUNDS["Ni_atom%"][1]) / 2
        individual[VAR_NAMES.index("M2_atom%")] = (BOUNDS["M2_atom%"][0] + BOUNDS["M2_atom%"][1]) / 2
        individual[VAR_NAMES.index("Milling_Time")] = (BOUNDS["Milling_Time"][0] + BOUNDS["Milling_Time"][1]) / 2
        individual[VAR_NAMES.index("Abs_Temp.")] = (BOUNDS["Abs_Temp."][0] + BOUNDS["Abs_Temp."][1]) / 2
        individual[VAR_NAMES.index("Abs_Pressure")] = (BOUNDS["Abs_Pressure"][0] + BOUNDS["Abs_Pressure"][1]) / 2
        individual[VAR_NAMES.index("Des_Temp.")] = (BOUNDS["Des_Temp."][0] + BOUNDS["Des_Temp."][1]) / 2
        individual[VAR_NAMES.index("Des_Pressure")] = (BOUNDS["Des_Pressure"][0] + BOUNDS["Des_Pressure"][1]) / 2
        known_materials.append(individual)
    
    for i in range(n_individuals):
        if i < len(known_materials):
            # Fixed: Use relative perturbation instead of multiplying by full reference
            variation = np.random.uniform(-0.03, 0.03, len(known_materials[0]))
            individual = known_materials[i % len(known_materials)] + variation * known_materials[i % len(known_materials)]
        else:
            # Random initialization within bounds
            individual = np.random.uniform(LOWER_BOUNDS, UPPER_BOUNDS)
            # Bias M2_AW toward known elements
            if np.random.random() > 0.6:
                individual[VAR_NAMES.index("M2_AW")] = np.random.choice([88.91, 138.91, 140.12, 140.91, 150.36])
        
        # For absorption-only scenario, fix desorption variables to mid-range
        # to reduce redundant search space
        if scenario_name == "Absorption_Only":
            individual[VAR_NAMES.index("Des_Temp.")] = (BOUNDS["Des_Temp."][0] + BOUNDS["Des_Temp."][1]) / 2
            individual[VAR_NAMES.index("Des_Pressure")] = (BOUNDS["Des_Pressure"][0] + BOUNDS["Des_Pressure"][1]) / 2
        
        # For desorption-only scenario, fix absorption variables to mid-range
        elif scenario_name == "Desorption_Only":
            individual[VAR_NAMES.index("Abs_Temp.")] = (BOUNDS["Abs_Temp."][0] + BOUNDS["Abs_Temp."][1]) / 2
            individual[VAR_NAMES.index("Abs_Pressure")] = (BOUNDS["Abs_Pressure"][0] + BOUNDS["Abs_Pressure"][1]) / 2
        
        individual = np.clip(individual, LOWER_BOUNDS, UPPER_BOUNDS)
        population.append(individual)
    
    return np.array(population)

class IsolatedProblem(Problem):
    """
    Multi-objective problem with selected objectives only.
    Proper constraint handling with constraint dominance.
    """
    
    N_CONSTRAINTS = 3
    CONSTRAINT_TOLERANCE = TOLERANCE_FRACTION

    def __init__(self, obj_indices: List[int], obj_directions: List[str], 
                 obj_names: List[str], models, x_scalers, y_scalers, 
                 model_feature_indices_list, scenario_name: str = "full"):
        
        self.models = models
        self.x_scalers = x_scalers
        self.y_scalers = y_scalers
        self.model_feature_indices_list = model_feature_indices_list
        self.obj_indices = obj_indices
        self.obj_directions = [obj_directions[i] for i in obj_indices]
        self.obj_names = [obj_names[i] for i in obj_indices]
        self.scenario_name = scenario_name
        
        super().__init__(
            n_var=len(VAR_NAMES),
            n_obj=len(obj_indices),
            n_constr=self.N_CONSTRAINTS,
            xl=LOWER_BOUNDS,
            xu=UPPER_BOUNDS,
            elementwise_evaluation=False
        )

    def _evaluate(self, X, out, *args, **kwargs):
        """Evaluate objectives and constraints"""
        # Predict all objectives
        all_preds = []
        for i in range(4):
            try:
                feature_indices = self.model_feature_indices_list[i]
                X_for_model = X[:, feature_indices]
                X_scaled = self.x_scalers[i].transform(X_for_model)
                y_scaled_pred = self.models[i].predict(X_scaled)
                y_scaled_pred = np.array(y_scaled_pred).reshape(-1, 1)
                
                try:
                    y_pred = self.y_scalers[i].inverse_transform(y_scaled_pred).flatten()
                except:
                    y_pred = y_scaled_pred.flatten()
                
                y_pred = np.clip(y_pred, 1e-6, None)
                
                # Convert based on direction
                if OBJECTIVE_DIRECTIONS[i] == "max":
                    all_preds.append(-y_pred)
                else:
                    all_preds.append(y_pred)
                    
            except Exception as e:
                print(f"Warning: Model {i} prediction failed: {e}")
                all_preds.append(np.ones(len(X)) * 1e10)
        
        # Select only requested objectives
        selected_preds = [all_preds[i] for i in self.obj_indices]
        out["F"] = np.column_stack(selected_preds)

        # Calculate constraints (violations, not just binary)
        v1 = constraint_structural_violation(X, tolerance=self.CONSTRAINT_TOLERANCE)
        v2 = constraint_pauling_violation(X, tolerance=self.CONSTRAINT_TOLERANCE)
        v3 = constraint_cohesive_violation(X, tolerance=self.CONSTRAINT_TOLERANCE)
        
        out["G"] = np.column_stack([v1, v2, v3])

def run_optimization_for_objectives(scenario_name: str, obj_indices: List[int], 
                                     obj_directions: List[str], obj_names: List[str],
                                     models, x_scalers, y_scalers, 
                                     model_feature_indices_list, run_number: int):
    """Run NSGA-II with full evolutionary operators"""
    
    run_seed = SEED_BASE + run_number
    
    print(f"\n{'='*60}")
    print(f"Scenario: {scenario_name}")
    print(f"Objectives: {[obj_names[i] for i in obj_indices]}")
    print(f"Population: {POP_SIZE} | Generations: {MAX_GEN} | Total evals: {POP_SIZE * MAX_GEN:,}")
    print(f"Run {run_number} (seed={run_seed})")
    print(f"{'='*60}")
    
    # Generate initial population with scenario-specific fixes
    initial_population = generate_feasible_population(POP_SIZE, scenario_name=scenario_name)
    
    # Create problem
    problem = IsolatedProblem(
        obj_indices=obj_indices,
        obj_directions=obj_directions,
        obj_names=obj_names,
        models=models,
        x_scalers=x_scalers,
        y_scalers=y_scalers,
        model_feature_indices_list=model_feature_indices_list,
        scenario_name=scenario_name
    )
    
    # NSGA2 with tournament selection (using default comparison)
    algorithm = NSGA2(
        pop_size=POP_SIZE,
        sampling=FloatRandomSampling(),
        crossover=SBX(prob=CROSSOVER_RATE, eta=CROSSOVER_ETA),
        mutation=PM(prob=MUTATION_RATE, eta=MUTATION_ETA),
        eliminate_duplicates=True
    )
    
    termination = get_termination("n_gen", MAX_GEN)
    
    # Initialize algorithm with custom population
    algorithm.setup(problem, seed=run_seed)
    pop = Population.new(X=initial_population)
    problem._evaluate(
        pop.get("X"),
        out={"F": np.zeros((len(pop), len(obj_indices))), "G": np.zeros((len(pop), 3))}
    )
    algorithm.pop = pop
    
    print("\n🚀 Running NSGA-II optimization...")
    print(f"   Crossover: SBX (p={CROSSOVER_RATE}, η={CROSSOVER_ETA})")
    print(f"   Mutation:  PM (p={MUTATION_RATE}, η={MUTATION_ETA})")
    print(f"   Selection: Binary Tournament (default NSGA-II)\n")
    
    res = minimize(problem, algorithm, termination, seed=run_seed, verbose=True)
    
    # Extract Pareto front
    if res.X is not None and len(res.X) > 0:
        final_X = res.X
        final_F = res.F
    elif hasattr(algorithm, 'pop') and algorithm.pop is not None:
        valid_inds = [ind for ind in algorithm.pop if ind.X is not None and ind.F is not None]
        if valid_inds:
            final_X = np.array([ind.X for ind in valid_inds])
            final_F = np.array([ind.F for ind in valid_inds])
        else:
            print("  ❌ No valid solutions found")
            return None, None, None
    else:
        print("  ❌ No solutions found")
        return None, None, None
    
    if len(final_X) == 0:
        print("  ❌ Empty solution set")
        return None, None, None
    
    # Get non-dominated front
    nds_idx = NonDominatedSorting().do(final_F, only_non_dominated_front=True)
    pareto_F = final_F[nds_idx].copy()
    pareto_X = final_X[nds_idx].copy()
    
    if len(pareto_X) == 0:
        print("  ❌ Empty Pareto front")
        return None, None, None
    
    # Convert back for maximization objectives
    reported_F = pareto_F.copy()
    for i, direction in enumerate([obj_directions[idx] for idx in obj_indices]):
        if direction == "max":
            reported_F[:, i] *= -1
    
    # Sort by first objective
    sort_idx = np.argsort(reported_F[:, 0], kind="stable")
    reported_F = reported_F[sort_idx]
    pareto_X = pareto_X[sort_idx]
    
    # Build DataFrame with ALL variables (for internal use)
    selected_obj_names = [obj_names[i] for i in obj_indices]
    df = pd.DataFrame(pareto_X, columns=VAR_NAMES)
    for i in range(len(selected_obj_names)):
        df[selected_obj_names[i]] = reported_F[:, i]
    
    # ========== FIX: Remove irrelevant variables from DataFrame based on scenario ==========
    if scenario_name == "Absorption_Only":
        # Remove desorption variables (they are fixed and irrelevant)
        irrelevant_cols = ["Des_Temp.", "Des_Pressure"]
        df = df.drop(columns=[col for col in irrelevant_cols if col in df.columns])
        print(f"  🧹 Removed irrelevant columns for Absorption_Only: {irrelevant_cols}")
    
    elif scenario_name == "Desorption_Only":
        # Remove absorption variables (they are fixed and irrelevant)
        irrelevant_cols = ["Abs_Temp.", "Abs_Pressure"]
        df = df.drop(columns=[col for col in irrelevant_cols if col in df.columns])
        print(f"  🧹 Removed irrelevant columns for Desorption_Only: {irrelevant_cols}")
    # ============================================================================
    
    # Calculate constraint violations
    v1 = constraint_structural_violation(pareto_X, tolerance=TOLERANCE_FRACTION)
    v2 = constraint_pauling_violation(pareto_X, tolerance=TOLERANCE_FRACTION)
    v3 = constraint_cohesive_violation(pareto_X, tolerance=TOLERANCE_FRACTION)
    
    df["C1_Violation"] = v1
    df["C2_Violation"] = v2
    df["C3_Violation"] = v3
    df["C1_Satisfied"] = v1 <= 1e-6
    df["C2_Satisfied"] = v2 <= 1e-6
    df["C3_Satisfied"] = v3 <= 1e-6
    df["All_Satisfied"] = df["C1_Satisfied"] & df["C2_Satisfied"] & df["C3_Satisfied"]
    df["Total_Violation"] = v1 + v2 + v3
    
    print(f"\n📊 Results:")
    print(f"  • Pareto solutions: {len(df)}")
    print(f"  • Constraint satisfaction:")
    print(f"      C1: {df['C1_Satisfied'].sum()}/{len(df)} ({100*df['C1_Satisfied'].sum()/len(df):.1f}%)")
    print(f"      C2: {df['C2_Satisfied'].sum()}/{len(df)} ({100*df['C2_Satisfied'].sum()/len(df):.1f}%)")
    print(f"      C3: {df['C3_Satisfied'].sum()}/{len(df)} ({100*df['C3_Satisfied'].sum()/len(df):.1f}%)")
    print(f"  • All constraints: {df['All_Satisfied'].sum()}/{len(df)} ({100*df['All_Satisfied'].sum()/len(df):.1f}%)")
    
    return df, pareto_X, reported_F
